        }

        doorState = DoorState.CLOSED;
    }

        setDoorState(DoorState.CLOSED);

package project.elevator;

import java.util.concurrent.PriorityBlockingQueue;

public class ElevatorController implements Runnable {
    private final PriorityBlockingQueue<Integer> upStops;
    private final PriorityBlockingQueue<Integer> downStops;
    private final Elevator elevator;
    private final Object monitor = new Object();

    public Elevator getElevator() {
        return elevator;
    }

    public boolean hasPendingStops() {
        return !upStops.isEmpty() || !downStops.isEmpty();
    }

    public ElevatorController(Elevator elevator) {
        this.elevator = elevator;
        upStops = new PriorityBlockingQueue<>();
        downStops = new PriorityBlockingQueue<>(10, (a, b) -> b - a);
    }

    public void addRequest(int destinationFloor) {
        synchronized (monitor) {
            int currentFloor = elevator.getCurrentFloor();

            if (currentFloor == destinationFloor) {
                return;
            }

            if (currentFloor < destinationFloor) {
                upStops.offer(destinationFloor);
            } else {
                downStops.offer(destinationFloor);
            }

            monitor.notifyAll();
        }
    }

    @Override
    public void run() {
        init();
    }

    private void init() {
        while (!Thread.currentThread().isInterrupted()) {
            synchronized (monitor) {
                try {
                    while (!hasPendingStops()) {
                        elevator.setDirection(Direction.IDLE);
                        System.out.println("Elevator Sleeping ... ");
                        monitor.wait();
                    }
                } catch (InterruptedException e) {
                    Thread.currentThread().interrupt();
                    return;
                }
            }

            while (!upStops.isEmpty()) {
                int destination = upStops.poll();
                elevator.setDirection(Direction.UP);
                elevator.move(destination);
            }

            while (!downStops.isEmpty()) {
                int destination = downStops.poll();
                elevator.setDirection(Direction.DOWN);
                elevator.move(destination);
            }
        }

    }
}

package project.elevator;

import java.util.List;

public class ElevatorScheduler {
    private final List<ElevatorController> elevatorControllers;
    private final ElevatorScheduleStrategy elevatorScheduleStrategy;

    public ElevatorScheduler(List<ElevatorController> elevatorControllers, ElevatorScheduleStrategy elevatorScheduleStrategy) {
        this.elevatorControllers = elevatorControllers;
        this.elevatorScheduleStrategy = elevatorScheduleStrategy;
    }

    public List<ElevatorController> getElevatorControllers() {
        return elevatorControllers;
    }

    public ElevatorScheduleStrategy getElevatorScheduleStrategy() {
        return elevatorScheduleStrategy;
    }

    public ElevatorController allocateElevator(int destinationFloor, Direction direction) {
        ElevatorController elevatorController = elevatorScheduleStrategy.select(elevatorControllers, destinationFloor, direction);
        elevatorController.addRequest(destinationFloor);
        return elevatorController;
    }
}

package project.elevator;

import java.util.List;

public class NearestScheduleStrategy implements ElevatorScheduleStrategy {
    @Override
    public ElevatorController select(List<ElevatorController> elevatorControllers, int requestedFloor, Direction direction) {

        int minDistance = Integer.MAX_VALUE;
        ElevatorController selectedController = null;


        // check for all elevators going up and down
        for (ElevatorController controller : elevatorControllers) {
            Elevator elevator = controller.getElevator();

            if (Direction.UP == direction && elevator.getDirection() == direction && requestedFloor > elevator.getCurrentFloor()) {
                int distance = Math.abs(elevator.getCurrentFloor() - requestedFloor);

                if (distance < minDistance) {
                    minDistance = distance;
                    selectedController = controller;
                }
            }

            if (Direction.DOWN == direction && elevator.getDirection() == direction && requestedFloor < elevator.getCurrentFloor()) {
                int distance = Math.abs(elevator.getCurrentFloor() - requestedFloor);

                if (distance < minDistance) {
                    minDistance = distance;
                    selectedController = controller;
                }
            }

        }

        // check for idle elevator
        if (selectedController == null) {
            for (ElevatorController controller : elevatorControllers) {
                int distance = Math.abs(controller.getElevator().getCurrentFloor() - requestedFloor);

                if (controller.getElevator().getDirection() == Direction.IDLE) {

                    if (distance < minDistance) {
                        minDistance = distance;
                        selectedController = controller;
                    }
                }

            }
        }

        // check for 3rd condition
        if (selectedController == null) {
            return elevatorControllers.get(0);
        }

        return selectedController;
    }
}
