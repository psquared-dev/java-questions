## 4. Board.java

```java
package project.tictactoe;

public class Board {
	private final PlayingPiece[][] matrix;

	public Board(int size) {
		this.matrix = new PlayingPiece[size][size];
	}

	public int getSize(){
		return matrix.length;
	}

	public boolean isPosFilled(int row, int col) {
		return matrix[row][col] != null;
	}

	public void fillPos(int row, int col, PlayingPiece pieceType) {
		matrix[row][col] = pieceType;
	}

	public PlayingPiece getPos(int row, int col) {
		return matrix[row][col];
	}

	public void display() {
		for (int i = 0; i < matrix.length; i++) {
			System.out.print(" | ");
			for (int j = 0; j < matrix[0].length; j++) {
				if (matrix[i][j] == null) {
					System.out.printf("%5s |", " ");
				} else {
					System.out.printf("%5s |", matrix[i][j]);
				}
			}
			System.out.print("\n--------------------------\n");
		}
	}
}
```

## 6. StandardWinningStretegy.java

```java
package project;

import project.tictactoe.Board;
import project.tictactoe.Player;
import project.tictactoe.PlayingPiece;

public class StandardWinningStretegy implements WinningStretegy {
	@Override
	public boolean checkWinner(Board board, Player player, int startRow, int startCol) {
		int desiredLength = board.getSize();
		int len;

		// check startRow
		len = 0;
		for (int col = 0; col < board.getSize(); col++) {
			if (player.getPlayingPiece().equals(board.getPos(startRow, col))) {
				len++;
			}
		}

		if (desiredLength == len) {
			return true;
		}

		// check startCol
		len = 0;
		for (int row = 0; row < board.getSize(); row++) {
			if (player.getPlayingPiece().equals(board.getPos(row, startCol))) {
				len++;
			}
		}

		if (desiredLength == len) {
			return true;
		}

		// check diagonal
		len = 0;
		if (startRow == startCol) {
			for (int i = 0; i < board.getSize(); i++) {
				if (player.getPlayingPiece().equals(board.getPos(i, i))) {
					len++;
				}
			}

			if (desiredLength == len) {
				return true;
			}
		}

		// check another diagonal
		len = 0;
		if (startRow + startCol == board.getSize() - 1) {
			for (int row = 0, col = board.getSize() - 1; row < board.getSize(); row++, col--) {
				if (player.getPlayingPiece().equals(board.getPos(row, col))) {
					len++;
				}
			}

			if (desiredLength == len) {
				return true;
			}
		}

		return false;
	}
}
```

## 7. Game.java

```java
package project.tictactoe;

import project.StandardWinningStretegy;
import project.WinningStretegy;

import java.util.LinkedList;
import java.util.Scanner;

public class Game {
	private final LinkedList<Player> queue;
	Board board;
	WinningStretegy winningStretegy;

	public Game(Board board) {
		this.queue = new LinkedList<>();
		this.board = board;
	}

	public void addPlayer(Player player) {
		this.queue.addLast(player);
	}

	public LinkedList<Player> getPlayers() {
		return queue;
	}

	public WinningStretegy getWinningStretegy() {
		return winningStretegy;
	}

	public void setWinningStretegy(WinningStretegy winningStretegy) {
		this.winningStretegy = winningStretegy;
	}

	public void play() {
		Scanner scanner = new Scanner(System.in);
		boolean shouldStop = false;
		int moves = 0;

		while (!shouldStop) {
			Player player = queue.pollFirst();

			System.out.print("[Player " + player.getName() + " ] Enter position: ");

			String input = scanner.nextLine();
			String[] split = input.split(",");

			int row = Integer.parseInt(split[0]);
			int col = Integer.parseInt(split[1]);

			if (row < 0 || row > board.getSize() - 1 || col < 0 || col > board.getSize() - 1) {
				System.out+.println("Invalid Position: [" + row + "," + col + "]");
				queue.addFirst(player);
			} else if (board.isPosFilled(row, col)) {
				System.out.println("Position: [" + row + "," + col + "]" + " is already filled");
				queue.addFirst(player);
			} else {
				board.fillPos(row, col, player.getPlayingPiece());
				queue.addLast(player);
				moves++;

				if (winningStretegy.checkWinner(board, player, row, col)) {
					shouldStop = true;
					System.out.println("Player: " + player.getName() + " won");
				} else if (moves == board.getSize() * board.getSize()) {
					shouldStop = true;
					System.out.println("Its a tie.");
				}
			}

			board.display();
		}
	}


	public static void main(String[] args) {
		Game game = new Game(new Board(3));

		game.setWinningStretegy(new StandardWinningStretegy());

		game.addPlayer(new Player("aa", new PlayingPiece(PieceType.O)));
		game.addPlayer(new Player("bb", new PlayingPiece(PieceType.X)));


		game.play();
	}
}
```
